    <script>
        document.addEventListener("DOMContentLoaded", function () {
            const contextPath = "${pageContext.request.contextPath}";
            const targetPath = "/apps9/EREHome/Erhm01View";

            const loginGovButton =
                document.getElementById("loginGovButton");

            const idMeButton =
                document.getElementById("idMeButton");

            const continueButton =
                document.getElementById("continueButton");

            function redirectToFIS(idp) {
                let redirectUrl =
                    contextPath +
                    "/FIS/app?target=" +
                    encodeURIComponent(targetPath);

                if (idp) {
                    redirectUrl =
                        contextPath +
                        "/FIS/app?idp=" +
                        encodeURIComponent(idp) +
                        "&target=" +
                        encodeURIComponent(targetPath);
                }

                window.location.href = redirectUrl;
            }

            if (loginGovButton) {
                loginGovButton.addEventListener(
                    "click",
                    function (event) {
                        event.preventDefault();
                        redirectToFIS("Login.gov");
                    }
                );
            }

            if (idMeButton) {
                idMeButton.addEventListener(
                    "click",
                    function (event) {
                        event.preventDefault();
                        redirectToFIS("idme");
                    }
                );
            }

            if (continueButton) {
                continueButton.addEventListener(
                    "click",
                    function (event) {
                        event.preventDefault();
                        redirectToFIS();
                    }
                );
            }
        });
    </script>
